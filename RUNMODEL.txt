#!/bin/zsh

monitor="${0:A:h}/scripts/monitor-feeds.zsh"

case "${1:-}" in
  -h|--help)
    cat <<'EOF'
Usage: zsh RUNMODEL.txt [--full-refresh]
       zsh RUNMODEL.txt hockey [--upload-only]
       zsh RUNMODEL.txt lol [--full-refresh] [--refresh-only]
       zsh RUNMODEL.txt hockey-rosters|hockey-data|hockey-odds
       zsh RUNMODEL.txt sync|sync-odds [--preview]

With no arguments, run and publish Hockey, then League of Legends.
With --full-refresh, also force the League of Legends historical rebuild.
Production runs include their existing Shiny deployments and Telegram updates.
EOF
    exit 0
    ;;
  hockey|lol|hockey-rosters|hockey-data|hockey-odds|sync|sync-odds)
    exec /bin/zsh "$monitor" "$@"
    ;;
  ''|--full-refresh)
    if (( $# > 1 )) || { (( $# == 1 )) && [[ "$1" != --full-refresh ]]; }; then
      print -u2 'Usage: zsh RUNMODEL.txt [--full-refresh]'
      exit 2
    fi
    ;;
  *)
    print -u2 "Unknown command: $1. See zsh RUNMODEL.txt --help."
    exit 2
    ;;
esac

print '=== Hockey ==='
/bin/zsh "$monitor" hockey
hockey_status=$?
(( hockey_status < 128 )) || exit "$hockey_status"

print '=== League of Legends ==='
/bin/zsh "$monitor" lol "$@"
lol_status=$?

printf 'Refresh exit codes: Hockey %s, League of Legends %s (0 = success)\n' "$hockey_status" "$lol_status"
(( lol_status == 0 )) || exit "$lol_status"
exit "$hockey_status"
